\begin{tikzpicture}[
  font=\small,
  >={Stealth[length=2.4mm]},
  box/.style   ={draw, rounded corners=2pt, align=center, inner sep=4pt, minimum height=9mm},
  wide/.style  ={box, text width=36mm},
  dec/.style   ={draw, diamond, aspect=2.1, align=center, inner sep=1pt},
  note/.style  ={draw, dashed, rounded corners=2pt, align=left, inner sep=5pt},
  lnk/.style   ={->, thick},
  group/.style ={draw, dashed, rounded corners=3pt, inner sep=7pt},
  neuron/.style={circle, draw, minimum size=4.6mm, inner sep=0pt},
]

%% ---------------- input chain ----------------
\node[wide] (split) {\textbf{Group-aware split}\\[1pt]
  all 20 $r$-points of\\ one case in the same fold};

\node[wide, right=9mm of split] (inp) {Process parameters\\ and coordinate\\[1pt]
  $\mathbf{x}=(Q,k,\alpha,\mu,v,r)$};

%% ---------------- shared trunk ----------------
\node[box, right=11mm of inp, text width=17mm] (fe)
  {Fourier\\ embedding\\ $\gamma(r)$, $k\!\le\!4$};

\foreach \l in {1,2,3}{
  \foreach \i in {1,2,3,4}{
    \node[neuron, fill=blue!25] (h\l\i) at ($(fe.east)+(8mm+\l*10mm, 10.5mm-\i*7mm)$) {};
  }
}
\foreach \i in {1,2,3,4}{
  \node[neuron, fill=green!45] (o\i) at ($(fe.east)+(46mm, 10.5mm-\i*7mm)$) {};
}
\node[right=0.8mm of o1, inner sep=1pt] (l1) {$\hat\sigma_{rr}$};
\node[right=0.8mm of o2, inner sep=1pt] (l2) {$\hat\sigma_{\theta\theta}$};
\node[right=0.8mm of o3, inner sep=1pt] (l3) {$\hat\sigma_{zz}$};
\node[right=0.8mm of o4, inner sep=1pt] (l4) {$\hat\tau_{rz}$};

\foreach \i in {1,2,3,4}{\draw[gray!55, thin] (fe.east) -- (h1\i);}
\foreach \l [evaluate=\l as \n using int(\l+1)] in {1,2}{
  \foreach \i in {1,2,3,4}{\foreach \j in {1,2,3,4}{\draw[gray!55, thin] (h\l\i) -- (h\n\j);}}}
\foreach \i in {1,2,3,4}{\foreach \j in {1,2,3,4}{\draw[gray!55, thin] (h3\i) -- (o\j);}}

\begin{scope}[on background layer]
  \node[group, fit=(fe)(h11)(h34)(l1)(l4),
        label={[yshift=-1pt]above:\textbf{Shared trunk:} 3 hidden layers $\times$ 96 $\tanh$}] (net) {};
\end{scope}

\draw[lnk] (split) -- (inp);
\draw[lnk] (inp) -- (fe);

%% ---------------- physics branch ----------------
\node[box, right=17mm of net, text width=34mm] (ad)
  {\textbf{Automatic}\\ \textbf{differentiation}\\[3pt]
   $\dfrac{\partial\hat\sigma_{ij}}{\partial r}$\;; the extended track\\ adds $\partial/\partial z$};
\draw[lnk] (net.east) -- (ad.west);

\node[box, below=7mm of ad, text width=52mm] (phys)
  {\textbf{Equilibrium residual}\\[2pt]
   $\displaystyle R_1=\frac{\partial\hat\sigma_{rr}}{\partial r}
      +\frac{\hat\sigma_{rr}-\hat\sigma_{\theta\theta}}{r}$\\[4pt]
   $\displaystyle \mathcal{L}_{\mathrm{phys}}
      =\bigl\langle \log\!\bigl(1+(R_1/s_{R_1})^2\bigr)\bigr\rangle$\\[2pt]
   {\footnotesize weak form: the same residual projected on $\sin(k\pi r)$, $k\le8$}};

\node[box, below=6mm of phys, text width=52mm] (bc)
  {\textbf{Traction-free boundary at $r=1$}\\[2pt]
   $\displaystyle \mathcal{L}_{\mathrm{bc}}=\Bigl\langle
     \bigl(\hat\sigma_{rr}/s_{rr}\bigr)^2
    +\bigl(\hat\tau_{rz}/s_{rz}\bigr)^2 \Bigr\rangle_{r=1}$};

\draw[lnk] (ad.south) -- (phys.north);
\draw[lnk] (phys.south) -- (bc.north);

\begin{scope}[on background layer]
  \node[group, fit=(ad)(phys)(bc),
        label={[yshift=-1pt]above:\textbf{Physics-informed terms}}] (physgrp) {};
\end{scope}

%% ---------------- data loss and total loss ----------------
\node[box, below=17mm of net.south, text width=62mm] (loss)
  {$\displaystyle \mathcal{L}_{\mathrm{data}}(\theta)=\frac{1}{N_d}\sum_{k=1}^{N_d}
    \bigl\|\hat{\boldsymbol\sigma}(\mathbf{x}_k;\theta)-\boldsymbol\sigma^{\mathrm{true}}(\mathbf{x}_k)\bigr\|_2^2$};
\draw[lnk] (net.south) -- (loss.north);

\node[box, below=7mm of loss, text width=76mm] (total)
  {$\displaystyle \mathcal{L}(\theta)=\mathcal{L}_{\mathrm{data}}
     +\lambda_{\mathrm{bc}}\,\mathcal{L}_{\mathrm{bc}}
     +\lambda_{\mathrm{phys}}\,g\,\mathcal{L}_{\mathrm{phys}}$,\quad
   $\lambda_{\mathrm{phys}}=0.3$, $\lambda_{\mathrm{bc}}=0.5$};
\draw[lnk] (loss) -- (total);
\draw[lnk] (bc.west) -- (total.east);

%% ---------------- optimiser loop ----------------
\node[dec, below=11mm of total, text width=34mm] (stop)
  {early stopping\\ or epoch limit};
\draw[lnk] (total) -- (stop);

\node[wide, left=16mm of stop, text width=40mm] (opt)
  {\textbf{AdamW}, no schedule\\[1pt]
   $\eta=3\cdot10^{-3}$, weight decay $10^{-4}$\\
   gradient clipping 1.0, batch 64};
\draw[lnk] (stop) -- node[above, pos=0.35, font=\scriptsize] {No} (opt);
\coordinate (back) at ($(net.south west)+(0,5mm)$);
\draw[lnk] (opt.west) -- ++(-9mm,0) |- (back);

\node[wide, right=16mm of stop, text width=34mm] (done)
  {\textbf{Trained surrogate}\\[1pt] best state restored};
\draw[lnk] (stop) -- node[above, pos=0.35, font=\scriptsize] {Yes} (done);

%% ---------------- fixed-configuration note ----------------
\node[note, below=9mm of opt.south west, anchor=north west, text width=140mm] (fix)
  {\textbf{Held fixed for every family and every split.}\ The architecture, the
   optimiser settings and the loss weights were set before any held-out region was
   evaluated and were not tuned per family or per split, so no search can leak a
   held-out region. The only quantity read from the data is the gradient-norm
   calibration $g=\lVert\nabla_\theta\mathcal{L}_{\mathrm{data}}\rVert_2/
   \lVert\nabla_\theta\mathcal{L}_{\mathrm{phys}}\rVert_2$, set once on the first
   mini-batch so that strong and weak forms enter with the same share of the data
   gradient ($g\approx0.165$ strong, $\approx4.97$ weak).};
